    low = t.lower()
    out = []
    for pat, name in SOLUTE_PATTERNS:
        if re.search(pat, low):
            out.append(name)
    return out

def find_scenario(t, fallback):
    for name, pat in SCEN_KW:
        if re.search(pat, t, re.I):
            return name
    return fallback

def extract_rules(chunk):
    raw = chunk["text"]
    # normalize digit-spacing artifacts from pypdf CJK extraction: "0 . 5%" -> "0.5%", "1 00%" -> "100%"
    t = re.sub(r"(\d)\s*\.\s*(\d)", r"\1.\2", raw)
    t = re.sub(r"(\d)\s+(\d{2}%|mg|g|mM|MPa|μM|µM)", r"\1\2", t)
    low = t.lower()
    # noise anchors: matches whose context is clearly not a formulation fact
    NOISE = ("琼脂糖凝胶", "切胶回收", "电泳", "不良反应发生率", "发生率", " gel ", "agarose")
    recs = []
    sols = find_solutes(t)
    scen = find_scenario(t, chunk.get("primary_theme", ""))
    concs = []
    for m in CONC_RANGE.finditer(t):
        concs.append((m.start(), m.group(0).strip()))
    for m in CONC_SINGLE.finditer(t):
        s = m.start()
        ctx = t[max(0, s - 60):s + 60].lower()
        if any(k in ctx for k in NOISE):
            continue
        concs.append((s, m.group(0).strip()))
    if RATIO.search(t):
        concs.append((RATIO.search(t).start(), RATIO.search(t).group(0)[:70]))
    effects = []
    for m in EFFECT.finditer(t):
        if any(k in m.group(0).lower() for k in NOISE):
            continue
        effects.append((m.start(), m.group(0).strip()[:80]))
    for m in EFFECT_NUM_FIRST.finditer(t):
        if any(k in m.group(0).lower() for k in NOISE):
            continue
        effects.append((m.start(), m.group(0).strip()[:80]))
    concs = concs[:6]
    effects = effects[:4]
    if not sols:
        return recs
    pairs = []
    if concs and effects:
        for c in concs[:3]:
            ce = min(effects, key=lambda e: abs(e[0] - c[0])) if effects else None
            pairs.append((c, ce))
    elif concs:
        for c in concs[:3]:
            pairs.append((c, None))
    elif effects:
        for e in effects[:2]:
            pairs.append((None, e))
    else:
        pairs.append((None, None))
    sol_near = lambda pos: (min(sols, key=lambda s: low.find(s.lower()) if low.find(s.lower()) >= 0 else 10**9)
                            if not pos else None)
    for (c, e) in pairs:
        pos = (c or e or (0, ""))[0]
        near = None
        if pos:
            best, bd = None, 10**9
            for s in sols:
                for mm in re.finditer(re.escape(s.lower()), low):
                    d = abs(mm.start() - pos)
                    if d < bd:
                        bd, best = d, s
            near = best if bd <= 400 else None
        recs.append({
            "solute_type": near or (sols[0] if len(sols) == 1 else "Multiple/Mixed"),
            "application_scenario": scen,
            "concentration_ratio": c[1] if c else None,
            "quantitative_effect": e[1] if e else None,
            "evidence": t[max(0, pos - 80):pos + 120].replace("\n", " ") if pos else t[:160].replace("\n", " "),
        })
    return recs

def extract_llm(chunk):
    from openai import OpenAI
    global _client
    try:
        _client
    except NameError:
        _client = OpenAI()
    resp = _client.chat.completions.create(
        model=LLM_MODEL,
        response_format={"type": "json_object"},
        temperature=0,
        messages=[{"role": "system", "content": LLM_PROMPT},
                  {"role": "user", "content": chunk["text"][:3500]}],
    )
    data = json.loads(resp.choices[0].message.content or "{}")
    out = []
    for r in data.get("records", [])[:4]:
        out.append({
            "solute_type": r.get("solute_type"),
            "application_scenario": r.get("application_scenario"),
            "concentration_ratio": r.get("concentration_ratio"),
            "quantitative_effect": r.get("quantitative_effect"),
            "evidence": (r.get("evidence") or "")[:160],
        })
    return out

def main():

    n_read = n_filtered = n_llm = 0
    llm_failed = 0
    use_llm = HAS_KEY

        for line in f:

            n_read += 1
            ic = c.get("info_class", "")

            n_filtered += 1
            recs = []
            if use_llm and n_llm < LLM_BUDGET:
                try:
                    recs = extract_llm(c)
                    n_llm += 1
                except Exception as e:
                    llm_failed += 1
                    if llm_failed >= 3:
                        use_llm = False
                    recs = extract_rules(c)
            else:
                recs = extract_rules(c)
            for r in recs:
                if not any([r.get("concentration_ratio"), r.get("quantitative_effect"),
                            r.get("solute_type")]):
                    continue
                key = (c["pub"], r.get("solute_type"), r.get("concentration_ratio"),
                       r.get("quantitative_effect"))
                if key in seen:
                    continue
                seen.add(key)
                rows.append({
                    "record_id": f"FD-{len(rows)+1:05d}",
                    "patent_id": c["pub"],
                    "chunk_id": c["chunk_id"],
                    "section": c.get("section", ""),
                    "info_class": ic,
                    "solute_type": r.get("solute_type") or "",
                    "application_scenario": r.get("application_scenario") or "",
                    "concentration_ratio": r.get("concentration_ratio") or "",
                    "quantitative_effect": r.get("quantitative_effect") or "",
                    "evidence_snippet": (r.get("evidence") or "")[:200],
                    "extraction_method": "llm" if (HAS_KEY and n_llm > 0 and not llm_failed) else "rule_based_v1",
                    "source_pdf": c.get("source_pdf", ""),
                })
    with open(OUT, "w", newline="", encoding="utf-8-sig") as f:
        w = csv.DictWriter(f, fieldnames=list(rows[0].keys()) if rows else
                           ["record_id", "patent_id", "chunk_id", "section", "info_class",
                            "solute_type", "application_scenario", "concentration_ratio",
                            "quantitative_effect", "evidence_snippet", "extraction_method", "source_pdf"])
        w.writeheader()
        w.writerows(rows)
    from collections import Counter
    print(f"[Task1] chunks_read={n_read} filtered(实施例/权利要求)={n_filtered} llm_calls={n_llm} llm_fail={llm_failed}")
    print(f"[Task1] formulation records={len(rows)} distinct patents={len({r['patent_id'] for r in rows})} elapsed={time.time()-t0:.0f}s")
    print(f"[Task1] solute_type top:", Counter(r['solute_type'] for r in rows).most_common(8))
    print(f"[Task1] scenario top:", Counter(r['application_scenario'] for r in rows).most_common(6))

if __name__ == "__main__":
    main()

def table_chunk(pub, page_no, tbl_idx, rows, ctx):
    m = meta.get(pub, {})
    clean = lambda s: (s or "").replace("|", "/").replace("\n", " ").strip()
    header = [clean(c) for c in rows[0]]
    body = [[clean(c) for c in r] for r in rows[1:]]
    ncol = max(1, len(header))
    md = ["| " + " | ".join(header + [""] * (ncol - len(header))) + " |",
          "|" + "---|" * ncol]
    for r in body[:60]:
        md.append("| " + " | ".join(r + [""] * (ncol - len(r))) + " |")
    return {
        "chunk_id": f"{pub}::table_p{page_no:03d}_{tbl_idx}",
        "chunk_type": "table",
        "patent_id": pub,
        "pub": pub,
        "page_number": page_no,
        "title": m.get("title", ""),
        "assignee": m.get("assignee", ""),
        "country": m.get("country", ""),
        "filing": m.get("filing", ""),
        "publication": m.get("publication", ""),
        "primary_theme": m.get("primary_theme", ""),
        "sha256_pdf": m.get("sha256_pdf", ""),
        "source_pdf": f"patents/pdf/{pub}.pdf",
        "table_content": "\n".join(md),
        "context_text": ctx,

    }

# -*- coding: utf-8 -*-

将中文/英文查询扩展为覆盖 中/英/日/德/法/西 语言等价术语的布尔检索式。

class CrossLingualQueryRewriter:
    """词典驱动的多语言布尔检索式重写器；可选 LLM 增补。"""

    def __init__(self, use_llm: bool = False, llm_model: str | None = None):
        self.use_llm = use_llm and bool(os.environ.get("OPENAI_API_KEY"))
        self.llm_model = llm_model or os.environ.get("OPENAI_MODEL", "gpt-4o-mini")
        self._extra_terms: list[str] = []

    # ------------------------------------------------------------- detection
    @staticmethod
    def _detect_core(query: str) -> bool:
        q = query.lower()
        zh = bool(ZH_DETECT.search(query))
        kana = bool(JA_KANA.search(query))
        en = any(t in q for t in ("compatible solute", "osmoprotectant", "osmolyte"))
        if zh:  # 中文查询中即使未出现核心词，也通过溶质词触发核心扩展
            return True
        return bool(en or kana or zh)

    def _detect_solutes(self, query: str) -> list[str]:
        q = query.lower()
        hits = []
        for canon, variants in SOLUTE_SYNONYMS.items():
            for v in variants:
                if v.lower() in q:
                    hits.append(canon)
                    break
        return hits

    def _detect_scenarios(self, query: str) -> list[str]:
        q = query.lower()
        hits = []
        for canon, variants in SCENARIO_SYNONYMS.items():
            for v in variants:
                if v.lower() in q:
                    hits.append(canon)
                    break
        return hits

    # ------------------------------------------------------------- builders
    @staticmethod
    def _or_group(terms: list[str]) -> str:
        seen, out = set(), []
        for t in terms:
            k = t.lower()
            if k not in seen:
                seen.add(k)
                out.append(f'"{t}"')
        return "(" + " OR ".join(out) + ")"

    def _core_query(self) -> str:
        all_core = [t for group in CORE_TERMS.values() for t in group]
        return self._or_group(all_core)

    def _solute_query(self, canon: str) -> str:
        return self._or_group(SOLUTE_SYNONYMS[canon])

    def _scenario_query(self, canon: str) -> str:
        return self._or_group(SCENARIO_SYNONYMS[canon])

    # ------------------------------------------------------------- LLM extras
    def _llm_expand(self, query: str) -> list[str]:
        try:
            from openai import OpenAI
            client = OpenAI()
            resp = client.chat.completions.create(
                model=self.llm_model,
                response_format={"type": "json_object"},
                temperature=0,
                messages=[
                    {"role": "system",
                     "content": "给出与查询等价的检索关键词扩展（中/英/日/德/法/西），"
                                "只输出JSON：{\"terms\":[...]}，最多10个，禁止解释。"},
                    {"role": "user", "content": query}],
            )
            import json
            return [str(t) for t in json.loads(resp.choices[0].message.content or "{}").get("terms", [])][:10]
        except Exception:
            return []

    # ------------------------------------------------------------- public API
    def rewrite(self, query: str) -> list[str]:
        """返回去重后的布尔检索式列表（核心式在前）。"""
        weighted = self.rewrite_weighted(query)
        seen, out = set(), []
        for q, _ in weighted:
            if q not in seen:
                seen.add(q)
                out.append(q)
        return out

    def rewrite_weighted(self, query: str) -> list[tuple[str, float]]:
        """返回 (布尔检索式, 权重) 列表，按权重降序。"""
        results: list[tuple[str, float]] = []
        core = self._detect_core(query)
        if core:
            results.append((self._core_query(), W_CORE))
        for canon in self._detect_solutes(query):
            results.append((self._solute_query(canon), W_SOLUTE))
        for canon in self._detect_scenarios(query):
            results.append((self._scenario_query(canon), W_SCEN))
        # 组合式: 核心 AND 溶质（若两者均命中）
        if core and self._detect_solutes(query):
            combo = self._core_query() + " AND " + self._solute_query(self._detect_solutes(query)[0])
            results.append((combo, min(1.0, W_CORE + 0.3 * W_SOLUTE)))
        if self.use_llm:
            for t in self._llm_expand(query):
                results.append((f'"{t}"', W_LLM))
        seen, out = set(), []
        for q, w in sorted(results, key=lambda x: -x[1]):
            if q in seen:
                continue
            seen.add(q)
            out.append((q, round(w, 2)))
        return out
